\begin{frame}[fragile]
    \frametitle{Analyseur lexical~: choix du plus long lexème possible}

    \begin{lstlisting}[language=c]
    var ++;
    var += 12;
    \end{lstlisting}

    \pause
    \vspace{2em}

    On choisira~:
    \begin{tikzpicture}[
        every node/.style={draw, rectangle},
        node distance=1mm and 1mm
    ]
        \node (variable1) {\verb|var|};
        \node (plusplus) [right=of variable1] {\verb|++|};
        \node (semicolon1) [right=of plusplus] {\verb|;|};
        \node (variable2) [right=of semicolon1] {\verb|var|};
        \node (pluseq) [right=of variable2] {\verb|+=|};
        \node (int) [right=of pluseq] {\verb|12|};
        \node (semicolon2) [right=of int] {\verb|;|};
    \end{tikzpicture}

    Et non~:
    \begin{tikzpicture}[
        every node/.style={draw, rectangle},
        node distance=1mm and 1mm
    ]
        \node (variable1) {\verb|v|};
        \node (variable2) [right=of variable1] {\verb|a|};
        \node (variable3) [right=of variable2] {\verb|r|};
        \node (plus1) [right=of variable3] {\verb|+|};
        \node (plus2) [right=of plus1] {\verb|+|};
        \node (semicolon1) [right=of plus2] {\verb|;|};
        \node (variable4) [right=of semicolon1] {\verb|v|};
        \node (variable5) [right=of variable4] {\verb|a|};
        \node (variable6) [right=of variable5] {\verb|r|};
        % \node (dots) [draw=none] {…}
        \node (plus3) [right=of variable6] {\verb|+|};
        \node (eq) [right=of plus3] {\verb|=|};
        \node (int1) [right=of eq] {\verb|1|};
        \node (int2) [right=of int1] {\verb|2|};
        \node (semicolon2) [right=of int2] {\verb|;|};
    \end{tikzpicture}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Grammaire~: exemple de la grammaire française}

    \[
    \begin{aligned}
        \text{Phrase} &\to \text{Sujet Verbe Complément} \\
        \pause
        \text{Sujet} &\to \text{Groupe nominal} \\
        \text{Sujet} &\to \text{Pronom personnel sujet} \\
        \pause
        \text{Complément} &\to \text{Groupe nominal} \\
        \pause
        \text{Groupe nominal} &\to \text{Déterminant Nom} \\
        \pause
        \text{Verbe} &\to v \\
        \text{Pronom personnel sujet} &\to p_s \\
        \text{Déterminant} &\to d \\
        \text{Nom} &\to n \\
    \end{aligned}
    \]
\end{frame}

\begin{frame}[fragile]
    \frametitle{Grammaire~: exemple de la grammaire française}

    Je mange une pomme

    \pause

    \[
    \begin{aligned}
        \text{Phrase} &\implies \text{Sujet Verbe Complément} \\
        \pause
                      &\implies \text{Pronom personnel sujet Verbe Complément} \\
        \pause
                      &\implies \text{\textit{Je} Verbe Complément} \\
        \pause
                      &\implies \text{\textit{Je mange} Complément} \\
        \pause
                      &\implies \text{\textit{Je mange} Groupe nominal} \\
        \pause
                      &\implies \text{\textit{Je mange} Déterminant Nom} \\
        \pause
                      &\implies \text{\textit{Je mange une} Nom} \\
        \pause
                      &\implies \text{\textit{Je mange une pomme}} \\
    \end{aligned}
    \]

\end{frame}

\begin{frame}[fragile]
    \frametitle{Grammaire~: exemple de la grammaire française}

    Le chat mange la souris

    \pause

    \[
    \begin{aligned}
        \text{Phrase} &\implies \text{Sujet Verbe Complément} \\
        \pause
                      &\implies \text{Groupe nominal Verbe Complément} \\
        \pause
                      &\implies \text{Déterminant Nom Verbe Complément} \\
        \pause
                      &\implies \text{\textit{Le} Nom Verbe Complément} \\
        \pause
                      &\implies \text{\textit{Le chat} Verbe Complément} \\
        \pause
                      &\implies \text{\textit{Le chat mange} Complément} \\
        \pause
                      &\implies \text{\textit{Le chat mange} Groupe nominal} \\
        \pause
                      &\implies \text{\textit{Le chat mange} Déterminant Nom} \\
        \pause
                      &\implies \text{\textit{Le chat mange la} Nom} \\
        \pause
                      &\implies \text{\textit{Le chat mange la souris}} \\
    \end{aligned}
    \]

\end{frame}
