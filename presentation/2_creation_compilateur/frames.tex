\subsubsection{LL(1)}
\begin{frame}[fragile]
    \frametitle{Algorithme LL(1)}

    \textit{Left to right, Leftmost derivation}

    Précalcul des premiers lexèmes de chaque règle

\end{frame}

\begin{frame}[fragile]
    \frametitle{Algorithme LL(1)~: les ensembles $premier(\dots)$}

    \begin{align*}
        S & \to +EE \\
        E & \to c   \\
        E & \to d
    \end{align*}

    \pause

    \begin{align*}
        premier(c)   & = \{c\}    \\
        premier(d)   & = \{d\}    \\
        premier(E)   & = \{c, d\} \\
        premier(+EE) & = \{+\}
    \end{align*}

\end{frame}

\begin{frame}[fragile]
    \frametitle{Algorithme LL(1)}

    \begin{algorithm}[H]
        \DontPrintSemicolon
        \Input{$\alpha$ un symbole, $texte$}
        \Output{arbre $\times$ texte}

        \Si{$texte$ est vide}{
            Erreur\;
        }
        \Si{$\alpha$ est un terminal}{
            \Si{$texte[0] = \alpha$}{
                \Return{
                    $\vcenter{\hbox{\begin{tikzpicture}
                                    \node[draw,rectangle]{$texte[0]$};
                                \end{tikzpicture}}}$,
                    $texte[1...]$
                }\;
            }
            \Sinon{
                Erreur\;
            }
        }


        \caption{LL(1)}
    \end{algorithm}

\end{frame}

\begin{frame}[fragile]
    \addtocounter{algocf}{-1}
    \frametitle{Algorithme LL(1)}

    \begin{algorithm}[H]
        \setcounter{AlgoLine}{7}
        \DontPrintSemicolon
        \tcc*[h]{On a $\alpha \to \overrightarrow{\beta_1}, \alpha\to\overrightarrow{\beta_2}, \dots$}

        \Si{$\nexists i / texte[0] \in premier(\overrightarrow{\beta_i})$}{
            Erreur\;
        }

        Posons $i$ tel que $texte[0] \in premier(\overrightarrow{\beta_i})$\;

        $ss\_arbre_1, texte_1 \gets LL1(\overrightarrow{\beta_i}[0], texte)$\;
        $ss\_arbre_2, texte_2 \gets LL1(\overrightarrow{\beta_i}[1], texte_1)$\;
        etc.\;

        \Return{
            $\vcenter{\hbox{\begin{tikzpicture}[
                                level distance=10mm,
                                level 1/.style={sibling distance=20mm},
                            ]
                            \node [draw, circle] {$\alpha$}
                            child {node[draw,rectangle]{$ss\_arbre_1$}}
                            child {node[draw,rectangle]{$ss\_arbre_2$}}
                            child {node {$\dots$}} ;
                        \end{tikzpicture}}}$,
            $texte_n$
        }\;

        \caption{LL(1)}
    \end{algorithm}

    \addtocounter{framenumber}{-1}
\end{frame}

\begin{frame}[fragile]
    \frametitle{Problème de LL(1)}

    \begin{align*}
        S & \to I                                                              \\
        S & \to E                                                              \\
        I & \to \text{if} \dots \text{then} \dots \text{end}                   \\
        E & \to \text{if} \dots \text{then} \dots \text{else} \dots \text{end}
    \end{align*}

    \pause

    \begin{align*}
        premier(I) & = \{\text{if}\} \\
        premier(E) & = \{\text{if}\}
    \end{align*}

    Ici, $premier(I) \cap premier(E) \ne \varnothing$~: la grammaire n’est
    \textbf{pas LL(1)}~!
\end{frame}
